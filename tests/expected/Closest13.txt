#include <stdio.h>

int main() {
    int a[100], i, k, n, b[100], x = 0;

    printf("Enter k: ");
    scanf("%d", &k);

    printf("Enter n: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++) {
        scanf("%d", &a[i]);
    }

    for (i = 0; i < n; i++) {
        b[i] = k - a[i];
    }

    for (i = 0; i < n; i++) {
        if (b[i] > b[i + 1]) {
            x = b[i];
        } else if (b[i] == b[i + 1]) {
            x = b[i];
        } else if (b[i] < b[i + 1]) {
            x = b[i + 1];
        }
    }

    printf("%d", x);

    return 0;
}
